def linear_search(arr, target): """선형 탐색: 처음부터 끝까지 순차적으로 비교한다.""" comparisons = 0 for i in range(len(arr)): comparisons += 1 print(f"Step {comparisons}: arr[{i}] = {arr[i]} 와 {target} 비교", end=" → ") if arr[i] == target: print(f"✅ 발견! 위치 {i}") return i, comparisons else: print("불일치") print(f"❌ {target}은(는) 리스트에 없다.") return -1, comparisons # ──────────────────────────────────────── # 🔧 [실험] 아래 두 줄의 값을 바꾸어 보자 # ──────────────────────────────────────── arr = [3, 8, 1, 15, 7, 12, 5, 9] target = 7 print(f"리스트: {arr}") print(f"찾을 값: {target}") print("=" * 50) position, comparisons = linear_search(arr, target) print("=" * 50) print(f"\n📊 총 비교 횟수: {comparisons}회") print(f"📊 리스트 크기 n = {len(arr)}, 최악의 경우 = {len(arr)}회")

이진 탐색은 정렬된 리스트에서만 동작한다. 매 단계마다 탐색 범위를 절반으로 줄이므로 시간 복잡도는 O(log n)이다. n = 1024일 때 최대 10회의 비교만으로 탐색이 완료된다.
import math def binary_search(arr, target): """이진 탐색: 정렬된 리스트에서 중앙값과 비교하여 범위를 절반씩 축소한다.""" lo, hi = 0, len(arr) - 1 comparisons = 0 while lo <= hi: mid = (lo + hi) // 2 comparisons += 1 print(f"Step {comparisons}: lo={lo}, hi={hi}, mid={mid}, arr[{mid}]={arr[mid]}", end=" → ") if arr[mid] == target: print(f"✅ 발견! 위치 {mid}") return mid, comparisons elif arr[mid] < target: print(f"{arr[mid]} < {target}, 오른쪽 범위로 이동") lo = mid + 1 else: print(f"{arr[mid]} > {target}, 왼쪽 범위로 이동") hi = mid - 1 print(f"❌ {target}은(는) 리스트에 없다.") return -1, comparisons # ──────────────────────────────────────── # 🔧 [실험] 아래 두 줄의 값을 바꾸어 보자 # ──────────────────────────────────────── arr = [2, 5, 8, 11, 14, 17, 20, 23, 26, 29] target = 17 print(f"정렬된 리스트: {arr}") print(f"찾을 값: {target}") print("=" * 50) position, comparisons = binary_search(arr, target) print("=" * 50) n = len(arr) print(f"\n📊 총 비교 횟수: {comparisons}회") print(f"📊 리스트 크기 n = {n}, 이론적 최대 = ⌈log₂({n})⌉ = {math.ceil(math.log2(n))}회") print(f"📊 선형 탐색 대비 효율: {n}회 → {comparisons}회 ({100*comparisons/n:.1f}%)")

버블 정렬은 인접한 두 원소를 비교하여 큰 값을 뒤로 밀어내는 방식이다. 한 번의 패스(pass)가 끝나면 가장 큰 값이 맨 뒤에 확정된다. 시간 복잡도는 O(n²)이며, 비교 횟수는 최대 n(n-1)/2회이다.
def bubble_sort(arr, verbose=True): """버블 정렬: 인접한 두 원소를 비교·교환하여 큰 값을 뒤로 보낸다.""" arr = arr.copy() n = len(arr) comparisons = 0 swaps = 0 for pass_num in range(n - 1): swapped_in_pass = False if verbose: print(f"\n--- Pass {pass_num + 1} ---") for j in range(n - 1 - pass_num): comparisons += 1 if arr[j] > arr[j + 1]: arr[j], arr[j + 1] = arr[j + 1], arr[j] swaps += 1 swapped_in_pass = True if verbose: print(f" arr[{j}]={arr[j+1]} > arr[{j+1}]={arr[j]} → 🔄 교환! {arr}") else: if verbose: print(f" arr[{j}]={arr[j]} ≤ arr[{j+1}]={arr[j+1]} → 유지") if verbose: print(f"Pass {pass_num + 1} 완료: {arr}") # 최적화: 한 번도 교환이 없으면 이미 정렬된 상태이다. if not swapped_in_pass: if verbose: print(f"⚡ 조기 종료! 더 이상 교환 없음.") break return arr, comparisons, swaps # ──────────────────────────────────────── # 🔧 [실험] 아래 리스트를 자유롭게 바꾸어 보자 # ──────────────────────────────────────── arr = [5, 3, 8, 1, 9, 2, 7] print(f"원본 리스트: {arr}") print("=" * 50) sorted_arr, comp, swp = bubble_sort(arr) print("=" * 50) n = len(arr) print(f"\n✅ 정렬 결과: {sorted_arr}") print(f"📊 총 비교: {comp}회, 총 교환: {swp}회") print(f"📊 최악 시간복잡도: n(n-1)/2 = {n*(n-1)//2}회")

삽입 정렬은 카드 게임에서 손에 든 카드를 정렬하는 방식과 동일하다. 정렬된 부분에 새 원소를 적절한 위치에 삽입(insert)한다. 평균 시간 복잡도는 O(n²)이지만, 거의 정렬된 입력에서는 O(n)에 가깝게 동작하는 적응적(adaptive) 알고리즘이다.
def insertion_sort(arr, verbose=True): """삽입 정렬: 정렬된 부분에 새 원소를 올바른 위치에 끼워 넣는다.""" arr = arr.copy() n = len(arr) comparisons = 0 for j in range(1, n): key = arr[j] i = j - 1 if verbose: print(f"\n--- j={j}, 삽입할 원소: {key} ---") print(f" 정렬된 부분: {arr[:j]}, 미정렬 부분: {arr[j:]}") # key보다 큰 원소들을 한 칸씩 뒤로 민다. while i >= 0 and arr[i] > key: comparisons += 1 if verbose: print(f" arr[{i}]={arr[i]} > {key} → 한 칸 뒤로 밀기") arr[i + 1] = arr[i] i -= 1 # 마지막 비교(밀지 않고 멈춘 비교)도 카운트한다. if i >= 0: comparisons += 1 arr[i + 1] = key if verbose: print(f" → ✅ {key}을(를) 위치 {i+1}에 삽입: {arr}") return arr, comparisons # ──────────────────────────────────────── # 🔧 [실험] 아래 리스트를 자유롭게 바꾸어 보자 # ──────────────────────────────────────── arr = [5, 3, 8, 1, 9, 2, 7] print(f"원본 리스트: {arr}") print("=" * 50) sorted_arr, comp = insertion_sort(arr) print("=" * 50) n = len(arr) print(f"\n✅ 정렬 결과: {sorted_arr}") print(f"📊 총 비교 횟수: {comp}회") print(f"📊 최악 시간복잡도: n(n-1)/2 = {n*(n-1)//2}회") print(f"📊 최선 시간복잡도(이미 정렬): n-1 = {n-1}회")

🐍 파이썬으로 직접 구현하기
욕심쟁이(greedy) 알고리즘은 매 단계에서 지역적 최적해를 선택하는 전략이다. 거스름돈 문제에서는 가장 큰 동전부터 최대한 많이 사용한다. 한국·미국 화폐 시스템에서는 항상 최적해를 보장하지만, 모든 경우에 최적은 아니다. 반례를 코드로 직접 확인해 본다.
def greedy_change(coins, amount, verbose=True): """욕심쟁이 알고리즘: 큰 동전부터 최대한 많이 사용한다.""" coins = sorted(coins, reverse=True) remaining = amount result = [] total_coins = 0 if verbose: print(f"💰 거스름돈: {amount}원") print(f"🪙 사용 가능 동전: {coins}원") print("=" * 50) for coin in coins: count = remaining // coin if count > 0: used = coin * count remaining -= used result.append((coin, count)) total_coins += count if verbose: print(f" {coin}원 × {count}개 = {used}원 사용 (남은 금액: {remaining}원)") else: if verbose: print(f" {coin}원: 사용 불가 (남은 금액 {remaining}원 < {coin}원)") if verbose: print("=" * 50) if remaining == 0: print(f"\n✅ 결과: 총 {total_coins}개 동전 사용") print(" " + " + ".join(f"{c}원×{n}" for c, n in result)) else: print(f"\n⚠️ 경고: {remaining}원을 거슬러 줄 수 없다!") return result, total_coins # ──────────────────────────────────────── # 🔧 [실험 1] 한국 화폐 시스템 (욕심쟁이 = 최적) # ──────────────────────────────────────── coins = [500, 100, 50, 10] amount = 760 result, total = greedy_change(coins, amount) # ──────────────────────────────────────── # 🔧 [실험 2] 욕심쟁이가 실패하는 반례 # ──────────────────────────────────────── print("\n\n" + "★" * 50) print("⚠️ 욕심쟁이가 최적을 보장하지 않는 경우") print("★" * 50) coins2 = [4, 3, 1] amount2 = 6 result2, total2 = greedy_change(coins2, amount2) print(f"\n🔍 비교:") print(f" - 욕심쟁이 결과: {total2}개 동전") print(f" - 진짜 최적해: 2개 동전 (3원 × 2)") print(f"\n💡 결론: 욕심쟁이는 화폐 단위가 '정준(canonical)'일 때만 최적이다.")

  • 한국 화폐 (전체): coins = [50000, 10000, 5000, 1000, 500, 100, 50, 10], amount = 87650
  • 미국 동전: coins = [25, 10, 5, 1], amount = 67 (cents) — 항상 최적이다.
  • 반례 발견: coins = [1, 5, 11], amount = 15 — 욕심쟁이는 5개(11+1+1+1+1), 최적은 3개(5+5+5)이다.
