```python
import numpy as np
from sklearn.decomposition import PCA

X = np.array([
    [2.5, 2.4], [0.5, 0.7], [2.2, 2.9], [1.9, 2.2],
    [3.1, 3.0], [2.3, 2.7], [2.0, 1.6], [1.0, 1.1],
    [1.5, 1.6], [1.1, 0.9],
])

# 1) 평균 중심화
X_centered = X - ___①___

# 2) 공분산행렬
Sigma = ___②___              # 힌트: np.cov(X, rowvar=False, ddof=1)

# 3) 고유분해 (대칭이므로 eigh 권장)
eigvals, eigvecs = ___③___(Sigma)

# 4) 큰 순서로 정렬
order = ___④___              # 힌트: np.argsort(eigvals)[::-1]
eigvals = eigvals[order]
eigvecs = eigvecs[:, order]

# 5) 분산 비율
ratio = ___⑤___ / ___⑥___    # 힌트: 각 고유값 / 전체 합

print("고유값:", eigvals)
print("분산 비율:", ratio)
```
